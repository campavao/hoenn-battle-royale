# ring-band: what the ROM's tile ring holds around the LCD (POK-319). The LCD is at BG
# VOFS 40 at rest, so ring rows 0..4 (tiles) are the 40 pixels the core draws above the
# picture and rows 25..31 the 56 below; every one of them should hold a tile. Since
# POK-329 the ring is 64 tile rows on 256x512 BGs (include/br/br_field.h, THE RING): a
# landing leaves the tile offsets at 0, so rows 32..55 are pos.y+16..pos.y+27 and rows
# 56..63 pos.y-4..pos.y-1, and they hold tiles too. The scroll registers are write-only
# on hardware, and the harness's bus read of them is open bus: the dump shows BGxCNT.
# Which map is where: BG2CNT names screen block 25 (0x0600C800), the middle layer a
# normal metatile's bottom half is drawn to; BG1CNT 27 (0xD800), the top layer over the
# sprites, 0 where no metatile has one; BG3CNT 29 (0xE800). pret's are 28, 29 and 30;
# BrField_InitRingBgs moves them (POK-329, vram-layout.txt). See the bgcnt dump.
wait 30
pokebytes gBrMailbox+0x2018 01 00 00 09 05 00 09 00 BD BB C7 FF
wait 400
say pos
dump *gSaveBlock1Ptr 4
say bgcnt-and-scroll
dump 0x04000008 24
say bg2-tilemap-row0
dump 0x0600C800 64
say bg2-tilemap-row4
dump 0x0600C900 64
say bg2-tilemap-row5-lcd-top
dump 0x0600C940 64
say bg2-tilemap-row24
dump 0x0600CE00 64
say bg2-tilemap-row25
dump 0x0600CE40 64
say bg2-tilemap-row30
dump 0x0600CF80 64
say bg2-tilemap-row31
dump 0x0600CFC0 64
say bg2-tilemap-row32
dump 0x0600D000 64
say bg2-tilemap-row55
dump 0x0600D5C0 64
say bg2-tilemap-row56
dump 0x0600D600 64
say bg2-tilemap-row63
dump 0x0600D7C0 64
say bg1-tilemap-row0
dump 0x0600D800 64
say bg1-tilemap-row31
dump 0x0600DFC0 64
say bg1-tilemap-row63
dump 0x0600E7C0 64
say bg3-tilemap-row0
dump 0x0600E800 64
say bg3-tilemap-row31
dump 0x0600EFC0 64
say bg3-tilemap-row63
dump 0x0600F7C0 64
# BG1..3 are 256x512 text BGs: BGxCNT bits 14..15 are 2.
expectbits u16 0x0400000A 0xC000 0x8000
expectbits u16 0x0400000C 0xC000 0x8000
expectbits u16 0x0400000E 0xC000 0x8000
# Column 0 of every ring row off the LCD, on the middle layer: this map's left edge is
# trees, so each is a tile, and a row the field never drew is 0.
expectne u16 0x0600C800 0
