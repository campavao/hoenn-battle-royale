# does the killing -- a ring centred far from Littleroot, left to bleed the party out.
wait 30

wait 400
expect u8 gPlayerPartyCount 1
# the eyeline only fires in a started match, and so does going out of one

# RING phase 1 centred on Fortree with r 1: Littleroot is outside, so the fog bleeds
pokebytes gBrMailbox+0x1018 0C 0A 07 00 00 00 01 19 0C 01 00 00

expect u8 gBrRing+5 1
# a level 5 Treecko bleeds out in about twenty ticks of 240 frames
wait 5200

# one ball for the one mon, plus the bag, both standing on this map
