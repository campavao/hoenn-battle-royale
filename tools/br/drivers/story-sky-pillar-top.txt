# story-sky-pillar-top: nobody sleeps at the top of SKY PILLAR but the story's RAYQUAZA
# (br_boot.c sets FLAG_HIDE_SKY_PILLAR_TOP_RAYQUAZA_STILL and FLAG_DEFEATED_RAYQUAZA).
#
# On SKY PILLAR TOP (24:85) at (14,9), two below where it sleeps at (14,7)... and the
# coord trigger at (14,9) that would wake it is already spent (VAR_SKY_PILLAR_RAYQUAZA_
# CRY_DONE). Flag 0x50 is byte 0x0A of the flags (*gSaveBlock1Ptr+0x127A), bit 0; the
# byte also holds RESCUED_BIRCH (0x52) and HIDE_CONTEST_POKE_BALL (0x56), both set at boot.
wait 30
poke u32 gBrMatch+8 8
pokebytes gBrMailbox+0x2018 81 00 18 55 0E 00 09 00 BD BB C7 FF
wait 400
poke u8 gBrMatch 2
expect u8 *gSaveBlock1Ptr+4 24
expect u8 *gSaveBlock1Ptr+5 85
shot top
expect u8 *gSaveBlock1Ptr+0x127A 0x45
say story sky pillar top ok
